if(!wilayaInput) { alert('يرجى كتابة اسم الولاية أولاً!'); return; } currentFileName = wilayaInput + ".json"; const name = document.getElementById('name').value.trim(); const phone = document.getElementById('phone').value.trim(); const profilePath = document.getElementById('profilePath').value.trim(); const menuPathsText = document.getElementById('menuPaths').value.trim(); const position = document.getElementById('position').value; const oldJsonText = document.getElementById('oldJson').value.trim(); if(!name || !phone) { alert('يرجى ملء اسم المطعم ورقم الهاتف على الأقل!'); return; } // تحويل أسطر صور المنيو إلى مصفوفة (Array) let menuImagesList = []; if (menuPathsText !== "") { menuImagesList = menuPathsText.split('\n').map(item => item.trim()).filter(item => item !== ""); } const newRest = { name: name, profileImg: profilePath, phone: phone, menuImages: menuImagesList }; let restaurantsList = []; if (oldJsonText !== "") { try { restaurantsList = JSON.parse(oldJsonText); if (!Array.isArray(restaurantsList)) restaurantsList = []; } catch (e) { alert('خطأ في تنسيق الكود القديم المكتوب! تأكد من صحته.'); return; } } if (position === 'start') { restaurantsList.unshift(newRest); } else { restaurantsList.push(newRest); } finalDataToDownload = JSON.stringify(restaurantsList, null, 4); const outputBox = document.getElementById('jsonOutput'); outputBox.value = finalDataToDownload; document.getElementById('outputBox').style.display = 'block'; } function downloadJSON() { const blob = new Blob([finalDataToDownload], { type: 'application/json' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = currentFileName; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); }
